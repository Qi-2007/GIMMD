cmake_minimum_required(VERSION 3.20)
project(gimmd LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "This prototype currently requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
    add_compile_options(/W4 /utf-8)
endif()
add_library(gimmd-core INTERFACE)
target_include_directories(gimmd-core INTERFACE src third_party/better_endfield)
add_executable(gimmd-vmd src/tools/vmd_inspect.cpp)
target_link_libraries(gimmd-vmd PRIVATE gimmd-core)
add_executable(gimmd-ik-inspect src/tools/ik_inspect.cpp)
target_link_libraries(gimmd-ik-inspect PRIVATE gimmd-core)
add_executable(gimmd-camera src/tools/camera_inspect.cpp)
target_link_libraries(gimmd-camera PRIVATE gimmd-core)
option(GIMMD_RUNTIME_PROBE "Build the pinned, bounded runtime camera observer" OFF)
option(GIMMD_POSE_ROUNDTRIP "Enable one-degree single-bone write/restore experiment" OFF)
option(GIMMD_SINGLE_TRACK_PLAYBACK "Enable bounded synthetic single-track playback" OFF)
option(GIMMD_FREEZE_TIME "Freeze and restore timeScale during the bounded single-track experiment" OFF)
option(GIMMD_ANIMATOR_PROBE "Observe the selected actor's Animator without writes" OFF)
option(GIMMD_ANIMATOR_TOGGLE "Disable/restore the selected Animator in one callback" OFF)
option(GIMMD_ANIMATOR_LEASE "Hold the selected Animator disabled during bounded single-track playback" OFF)
option(GIMMD_MULTI_TRACK_PLAYBACK "Enable a bounded ten-bone relative VMD preview" OFF)
option(GIMMD_PLAYER_SESSION "Enable timeline controls and PCM audio in the multi-track preview" OFF)
option(GIMMD_BIND_POSE_PROBE "Observe Avatar reference-skeleton interfaces" OFF)
option(GIMMD_FULL_BODY_FK "Drive 52 target bones using verified rest axes and canonical source stance" OFF)
option(GIMMD_ROOT_MOTION "Apply anchored MMD Center translation to the visual skeleton root" OFF)
option(GIMMD_LEG_IK "Solve MMD foot IK with independent two-bone leg chains" OFF)
option(GIMMD_TOE_IK "Aim the ankle toward MMD child toe IK targets" OFF)
option(GIMMD_CAMERA_PLAYBACK "Play camera VMD synchronized to the player timeline" OFF)
if(GIMMD_CAMERA_PLAYBACK AND (NOT GIMMD_PLAYER_SESSION OR GIMMD_CAMERA_ROUNDTRIP))
    message(FATAL_ERROR "Camera playback requires player session and excludes camera roundtrip")
endif()
option(GIMMD_CAMERA_ROUNDTRIP "Verify and restore a small camera position/rotation write within one callback" OFF)
if(GIMMD_CAMERA_ROUNDTRIP AND (NOT GIMMD_RUNTIME_PROBE OR GIMMD_SINGLE_TRACK_PLAYBACK OR GIMMD_POSE_ROUNDTRIP OR GIMMD_ANIMATOR_TOGGLE))
    message(FATAL_ERROR "Camera roundtrip requires a read-only runtime observer build")
endif()
if(GIMMD_TOE_IK AND NOT GIMMD_LEG_IK)
    message(FATAL_ERROR "GIMMD_TOE_IK requires leg IK")
endif()
if(GIMMD_LEG_IK AND NOT GIMMD_ROOT_MOTION)
    message(FATAL_ERROR "GIMMD_LEG_IK requires root motion")
endif()
if(GIMMD_ROOT_MOTION AND NOT GIMMD_FULL_BODY_FK)
    message(FATAL_ERROR "GIMMD_ROOT_MOTION requires full-body FK")
endif()
if(GIMMD_FULL_BODY_FK AND NOT GIMMD_MULTI_TRACK_PLAYBACK)
    message(FATAL_ERROR "GIMMD_FULL_BODY_FK requires multi-track playback")
endif()
if(GIMMD_BIND_POSE_PROBE AND NOT (GIMMD_ANIMATOR_PROBE OR GIMMD_ANIMATOR_LEASE))
    message(FATAL_ERROR "GIMMD_BIND_POSE_PROBE requires Animator observation")
endif()
if(GIMMD_PLAYER_SESSION AND NOT GIMMD_MULTI_TRACK_PLAYBACK)
    message(FATAL_ERROR "GIMMD_PLAYER_SESSION requires GIMMD_MULTI_TRACK_PLAYBACK")
endif()
if(GIMMD_MULTI_TRACK_PLAYBACK AND NOT GIMMD_ANIMATOR_LEASE)
    message(FATAL_ERROR "GIMMD_MULTI_TRACK_PLAYBACK requires GIMMD_ANIMATOR_LEASE")
endif()
if(GIMMD_ANIMATOR_LEASE AND (NOT GIMMD_SINGLE_TRACK_PLAYBACK OR GIMMD_FREEZE_TIME OR GIMMD_ANIMATOR_TOGGLE))
    message(FATAL_ERROR "GIMMD_ANIMATOR_LEASE requires single-track playback and excludes freeze-time / toggle")
endif()
if(GIMMD_ANIMATOR_TOGGLE AND NOT GIMMD_ANIMATOR_PROBE)
    message(FATAL_ERROR "GIMMD_ANIMATOR_TOGGLE requires GIMMD_ANIMATOR_PROBE")
endif()
if(GIMMD_ANIMATOR_PROBE AND (NOT GIMMD_RUNTIME_PROBE OR (GIMMD_SINGLE_TRACK_PLAYBACK AND NOT GIMMD_ANIMATOR_LEASE) OR GIMMD_POSE_ROUNDTRIP))
    message(FATAL_ERROR "GIMMD_ANIMATOR_PROBE requires a read-only runtime probe")
endif()
if(GIMMD_FREEZE_TIME AND NOT GIMMD_SINGLE_TRACK_PLAYBACK)
    message(FATAL_ERROR "GIMMD_FREEZE_TIME requires GIMMD_SINGLE_TRACK_PLAYBACK")
endif()
if(GIMMD_SINGLE_TRACK_PLAYBACK AND (NOT GIMMD_RUNTIME_PROBE OR GIMMD_POSE_ROUNDTRIP))
    message(FATAL_ERROR "Single-track playback requires runtime probe and excludes pose roundtrip")
endif()
if(GIMMD_POSE_ROUNDTRIP AND NOT GIMMD_RUNTIME_PROBE)
    message(FATAL_ERROR "GIMMD_POSE_ROUNDTRIP requires GIMMD_RUNTIME_PROBE")
endif()
if(GIMMD_RUNTIME_PROBE)
    add_library(gimmd-minhook STATIC
        third_party/minhook/src/buffer.c
        third_party/minhook/src/hook.c
        third_party/minhook/src/trampoline.c
        third_party/minhook/src/hde/hde64.c)
    target_include_directories(gimmd-minhook PUBLIC third_party/minhook/include)
    add_library(gimmd-probe SHARED src/windows/runtime_probe.cpp)
    target_include_directories(gimmd-probe PRIVATE src third_party/better_endfield)
    target_link_libraries(gimmd-probe PRIVATE gimmd-minhook bcrypt user32)
    add_executable(gimmd-runtime-tests src/windows/runtime_probe.cpp)
    target_compile_definitions(gimmd-runtime-tests PRIVATE GIMMD_OBSERVER_TEST)
    target_include_directories(gimmd-runtime-tests PRIVATE src third_party/better_endfield)
    target_link_libraries(gimmd-runtime-tests PRIVATE gimmd-minhook bcrypt user32)
    if(GIMMD_POSE_ROUNDTRIP)
        target_compile_definitions(gimmd-probe PRIVATE GIMMD_POSE_ROUNDTRIP)
        target_compile_definitions(gimmd-runtime-tests PRIVATE GIMMD_POSE_ROUNDTRIP)
    endif()
    if(GIMMD_SINGLE_TRACK_PLAYBACK)
        target_compile_definitions(gimmd-probe PRIVATE GIMMD_SINGLE_TRACK_PLAYBACK)
        target_compile_definitions(gimmd-runtime-tests PRIVATE GIMMD_SINGLE_TRACK_PLAYBACK)
    endif()
    if(GIMMD_FREEZE_TIME)
        target_compile_definitions(gimmd-probe PRIVATE GIMMD_FREEZE_TIME)
        target_compile_definitions(gimmd-runtime-tests PRIVATE GIMMD_FREEZE_TIME)
    endif()
    if(GIMMD_ANIMATOR_PROBE OR GIMMD_ANIMATOR_LEASE)
        target_compile_definitions(gimmd-probe PRIVATE GIMMD_ANIMATOR_PROBE)
        target_compile_definitions(gimmd-runtime-tests PRIVATE GIMMD_ANIMATOR_PROBE)
    endif()
    if(GIMMD_ANIMATOR_LEASE)
        target_compile_definitions(gimmd-probe PRIVATE GIMMD_ANIMATOR_LEASE)
        target_compile_definitions(gimmd-runtime-tests PRIVATE GIMMD_ANIMATOR_LEASE)
    endif()
    if(GIMMD_MULTI_TRACK_PLAYBACK)
        target_compile_definitions(gimmd-probe PRIVATE GIMMD_MULTI_TRACK_PLAYBACK)
        target_compile_definitions(gimmd-runtime-tests PRIVATE GIMMD_MULTI_TRACK_PLAYBACK)
    endif()
    if(GIMMD_PLAYER_SESSION)
        target_compile_definitions(gimmd-probe PRIVATE GIMMD_PLAYER_SESSION)
        target_compile_definitions(gimmd-runtime-tests PRIVATE GIMMD_PLAYER_SESSION)
    endif()
    if(GIMMD_BIND_POSE_PROBE)
        target_compile_definitions(gimmd-probe PRIVATE GIMMD_BIND_POSE_PROBE)
        target_compile_definitions(gimmd-runtime-tests PRIVATE GIMMD_BIND_POSE_PROBE)
    endif()
    if(GIMMD_FULL_BODY_FK)
        target_compile_definitions(gimmd-probe PRIVATE GIMMD_FULL_BODY_FK GIMMD_BIND_POSE_PROBE)
        target_compile_definitions(gimmd-runtime-tests PRIVATE GIMMD_FULL_BODY_FK GIMMD_BIND_POSE_PROBE)
    endif()
    if(GIMMD_ROOT_MOTION)
        target_compile_definitions(gimmd-probe PRIVATE GIMMD_ROOT_MOTION)
        target_compile_definitions(gimmd-runtime-tests PRIVATE GIMMD_ROOT_MOTION)
    endif()
    if(GIMMD_LEG_IK)
        target_compile_definitions(gimmd-probe PRIVATE GIMMD_LEG_IK)
        target_compile_definitions(gimmd-runtime-tests PRIVATE GIMMD_LEG_IK)
    endif()
    if(GIMMD_TOE_IK)
        target_compile_definitions(gimmd-probe PRIVATE GIMMD_TOE_IK)
        target_compile_definitions(gimmd-runtime-tests PRIVATE GIMMD_TOE_IK)
    endif()
    if(GIMMD_CAMERA_PLAYBACK)
        target_compile_definitions(gimmd-probe PRIVATE GIMMD_CAMERA_PLAYBACK)
        target_compile_definitions(gimmd-runtime-tests PRIVATE GIMMD_CAMERA_PLAYBACK)
    endif()
    if(GIMMD_CAMERA_ROUNDTRIP)
        target_compile_definitions(gimmd-probe PRIVATE GIMMD_CAMERA_ROUNDTRIP)
        target_compile_definitions(gimmd-runtime-tests PRIVATE GIMMD_CAMERA_ROUNDTRIP)
    endif()
    target_link_libraries(gimmd-probe PRIVATE winmm)
    target_link_libraries(gimmd-runtime-tests PRIVATE winmm)
    if(GIMMD_ANIMATOR_TOGGLE)
        target_compile_definitions(gimmd-probe PRIVATE GIMMD_ANIMATOR_TOGGLE)
        target_compile_definitions(gimmd-runtime-tests PRIVATE GIMMD_ANIMATOR_TOGGLE)
    endif()
else()
    add_library(gimmd-probe SHARED src/windows/probe.cpp)
endif()
add_executable(gimmd-probe-host src/windows/probe_host.cpp)
add_executable(gimmd-inject-probe src/windows/inject_probe.cpp)
add_executable(gimmd-tests tests/core_tests.cpp)
target_link_libraries(gimmd-tests PRIVATE gimmd-core)
enable_testing()
add_test(NAME core COMMAND gimmd-tests)
if(GIMMD_RUNTIME_PROBE)
    add_test(NAME runtime-observer COMMAND gimmd-runtime-tests)
endif()
